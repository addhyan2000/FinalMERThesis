\begin{description}

	% --- corpora and datasets ---
	\item[CASME II] Chinese Academy of Sciences Micro-Expression database II
	\item[CAS(ME)$^2$] Chinese Academy of Sciences Macro- and Micro-Expression database
	\item[CIFAR] Canadian Institute for Advanced Research (dataset family)
	\item[CK+] Extended Cohn--Kanade dataset
	\item[JAFFE] Japanese Female Facial Expression database
	\item[MMEW] Micro-and-Macro Expression Warehouse
	\item[MMI] MMI Facial Expression Database
	\item[MUG] Multimedia Understanding Group facial expression database
	\item[Multi-PIE] Multi Pose, Illumination and Expression database
	\item[SAMM] Spontaneous Actions and Micro-Movements dataset
	\item[SMIC] Spontaneous Micro-Expression Corpus
	\item[USF-HD] University of South Florida High-Definition dataset

	% --- benchmarks and evaluation protocols ---
	\item[CDE] Composite Database Evaluation
	\item[LOSO] Leave-one-subject-out (cross-validation)
	\item[MEGC] Micro-Expression Grand Challenge
	\item[SDE] Sole Database Evaluation

	% --- metrics ---
	\item[F1] Harmonic mean of precision and recall
	\item[FN] False negative
	\item[FP] False positive
	\item[TP] True positive
	\item[UAR] Unweighted average recall
	\item[UF1] Unweighted F1

	% --- the phenomenon, annotation and capture ---
	\item[AU] Action Unit (plural AUs)
	\item[FACS] Facial Action Coding System
	\item[fps] Frames per second
	\item[ME] Micro-expression (plural MEs)
	\item[SD] Standard deviation

	% --- hand-crafted features and preprocessing ---
	\item[AMM] Amplitude-based motion magnification
	\item[ASM] Active Shape Model
	\item[Bi-WOOF] Bi-Weighted Oriented Optical Flow
	\item[EVM] Eulerian Video Magnification
	\item[FFT] Fast Fourier transform
	\item[HIGO] Histogram of Image Gradient Orientation
	\item[HOG] Histogram of Oriented Gradients
	\item[JCFDA] Joint cascade face detection and alignment
	\item[LBP] Local Binary Patterns
	\item[LBP-TOP] Local Binary Patterns on Three Orthogonal Planes
	\item[OFF] Optical flow features
	\item[OFW] Optical flow weighted features
	\item[OSF] Optical strain features
	\item[OSW] Optical strain weighted features
	\item[PMM] Phase-based motion magnification
	\item[TIM] Temporal Interpolation Model (TIM10: interpolated to ten frames)
	\item[TV-L1] Total-variation regularisation with an L1 data term

	% --- network architectures and modules ---
	\item[C3D] Three-dimensional convolutional network
	\item[CBAM] Convolutional Block Attention Module
	\item[CNN] Convolutional neural network
	\item[DenseNet] Densely Connected Convolutional Network
	\item[ECA] Efficient Channel Attention
	\item[EMME] Extended Magnified Micro-Expression database
	\item[EMR] Expression Magnification and Reduction
	\item[GC] Global Context (attention module)
	\item[LSTM] Long short-term memory
	\item[OFF-ApexNet] Optical Flow Features from Apex frame Network
	\item[RCN] Recurrent convolutional network
	\item[ResNet] Residual Network
	\item[SE] Squeeze-and-Excitation
	\item[SimAM] Simple, parameter-free attention module
	\item[SLSTT] Short and Long range relation based Spatio-Temporal Transformer
	\item[SRM] Style-based Recalibration Module
	\item[STRCN] Spatiotemporal Recurrent Convolutional Network
	\item[STSTNet] Shallow Triple Stream Three-dimensional CNN
	\item[VGG-Face] VGG face-recognition network (VGGFace2: its successor dataset and model)
	\item[ViT] Vision Transformer
	\item[XBM] Cross-Batch Memory

	% --- training, optimisation and numerics ---
	\item[AdamW] Adam optimiser with decoupled weight decay
	\item[AMP] Automatic mixed precision
	\item[BN] Batch normalisation
	\item[FL] Focal loss
	\item[GELU] Gaussian error linear unit
	\item[Grad-CAM++] Generalised gradient-based class activation mapping
	\item[NaN] Not a Number (invalid floating-point value)
	\item[RBF] Radial basis function
	\item[ReLU] Rectified linear unit
	\item[SVM] Support vector machine

	% --- computing environment ---
	\item[CPU] Central processing unit
	\item[CUDA] Compute Unified Device Architecture
	\item[GPU] Graphics processing unit
	\item[RAM] Random-access memory
	\item[VRAM] Video memory (on-device GPU memory)
